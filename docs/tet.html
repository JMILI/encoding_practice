UDP 协议位域教程

一句话：网卡按「先到的字节放低地址」把大端字节流填进缓冲区； 按字节读不受端序影响，按 uint16_t / uint32_t 读会受主机端序影响， 因此解析多字节字段前要先 ntoh，再按声明顺序读位域。

阅读顺序（建议从上到下）

  4. data2 图解 — 4 字节位域

  7. C++ 协议结构体 / 收发包 — protocol.h + parse + build

§1 网络字节序的接收与处理流程

1.1 线上字节怎么进来？接到哪里？

UDP 载荷是一串按发送顺序排列的字节。接收侧（socket / DMA）把它们写入一块连续缓冲区： 先到达的字节落在低地址，后到达的落在高地址。

① 网络：对端按「字节0、字节1、字节2…」顺序发出（协议约定多字节整数用大端：高字节先发）。
② 网卡 / 协议栈：按到达顺序写入 uint8_t buf[]：buf[0] = 第一个字节（低地址），buf[n-1] = 最后一个字节（高地址）。
③ 临时存储：此时缓冲区里只是「原始字节阵列」，还没有「uint16 等于多少」的语义——语义取决于你怎么读。
端序何时介入？ 写入 buf[] 的过程与本机是大端还是小端无关——都是低地址先填先到的字节。 端序只在「把多个字节解释成一个整数」或「把整数写回多个字节」时出现。

1.2 按字节读 —— 大小端结果相同

每次只读 1 个 uint8_t，不存在「字节谁在前」的歧义：

C++read_by_byte.cpp
// 假设线上发来 4 字节（大端表示整数 0x12345678）：
//   先发 0x12，再发 0x34，再发 0x56，再发 0x78
// 接收后缓冲区低地址 → 高地址：
uint8_t buf[4] = {0x12, 0x34, 0x56, 0x78};

// 按「单个字节」下标读取：大端机、小端机结果完全一样
// 因为根本没有把多字节「合成一个整数」，不存在端序解释问题
printf("%02X %02X %02X %02X\n", buf[0], buf[1], buf[2], buf[3]);
// 输出 → 12 34 56 78

1.3 按基本类型读 —— 大小端结果不同

若把缓冲区的 4 个字节拷进 uint32_t，CPU 会按本机端序解释内存里的字节排列：

C++read_as_uint32.cpp
uint8_t buf[4] = {0x12, 0x34, 0x56, 0x78};

uint32_t v = 0;
memcpy(&v, buf, 4);   // 把 4 字节「原样」拷进 v 的内存映像

// 小端机：整数的「低字节」放在低地址
//   内存：12 34 56 78  →  数值被解释为 0x78563412
// 大端机：整数的「高字节」放在低地址
//   内存：12 34 56 78  →  数值被解释为 0x12345678
//
// 所以：同样的线上字节，按 uint32_t 直接读，两端结果不同！
// 正确做法：先 ntohl(v)（或按大端手工拼），得到协议规定的逻辑值 0x12345678
读法小端主机得到的数值大端主机得到的数值
按字节 buf[i] 12,34,56,78（相同） 12,34,56,78（相同）
memcpy 进 uint32_t 后直接当主机值用 0x78563412 0x12345678
ntohl 之后 0x12345678 0x12345678
小结：接收 = 低地址→高地址填 buf（无端序）； 按字节读两端一致； 按 2/4 字节类型读必须先 ntoh，再读位域（见 §7）。

§2 报文长什么样

UDP 单播载荷按网络字节序（大端）依次发送（与交互实验台默认数据一致）：

type (1B) | len (2B) | fc (变长) | data1 (2B) | data2 (4B) | data3 (1B) | data4 (3B)
  • len = fc + data1(2) + data2(4) + data3(1) + data4(3)；示例 fc=3 → len=13 (0x000D)
  • fc：每字节最低位为续位（1=还有下一字节，0=结束）
  • data3：1 字节位域 lo[2:0] / hi[7:3]；data4：3 字节跨字节位域
示例：11 | 00 0D | 51 73 B8 | 55 B5 | 93 65 34 12 | AD | AB 55 55（共 16 字节）

§7 C++ 协议结构体与收发包

核心思路：位域声明顺序 = 内存低位→高位的排布； 收包只要把线上字节 memcpy 进总包结构体，再 ntoh() 转字节序， 就可以直接读 pkt.data1.code 等字段——不必再手工移位拆位。

约定：#pragma pack(push, 1) 一字节对齐；位域谁先写谁占 bit0。 本图解示例报文 fc 固定 3 字节，故总包可定长；实战若 fc 变长，先扫续位得到偏移，再对「定长尾」做同样的 memcpy + ntoh。

文件 1 — 协议结构体（所有字段 + 总包）

C++protocol.h
#pragma once
#include <cstdint>
#include <cstring>
#include <arpa/inet.h>   // ntohs / ntohl / htons / htonl

#pragma pack(push, 1)     // 1 字节对齐：无 padding，布局与线上字节一一对应

// ---------- data1：2 字节位域 ----------
// 声明顺序 = bit0 起：code(10) → S3 → S2 → S1 → typeCode(3)
struct Data1 {
    unsigned int code     : 10;  // bit0  ~ 9
    unsigned int S3       : 1;   // bit10
    unsigned int S2       : 1;   // bit11
    unsigned int S1       : 1;   // bit12
    unsigned int typeCode : 3;   // bit13 ~ 15

    void ntoh() {                // 网络大端 → 主机序（收包后调用）
        uint16_t v;
        memcpy(&v, this, 2);     // 先把 2 字节内存映像当成整数
        v = ntohs(v);            // 小端机交换高低字节；大端机为空操作
        memcpy(this, &v, 2);     // 写回后，位域成员语义才正确
    }
    void hton() {                // 主机序 → 网络大端（发送前调用）
        uint16_t v;
        memcpy(&v, this, 2);
        v = htons(v);
        memcpy(this, &v, 2);
    }
};

// ---------- data2：4 字节位域 ----------
struct Data2 {
    unsigned int feild_1 : 4;    // bit0  ~ 3
    unsigned int feild_2 : 4;    // bit4  ~ 7
    unsigned int feild_3 : 4;    // bit8  ~ 11
    unsigned int feild_4 : 4;    // bit12 ~ 15
    unsigned int sysCode : 16;   // bit16 ~ 31

    void ntoh() {
        uint32_t v;
        memcpy(&v, this, 4);
        v = ntohl(v);            // 4 字节端序转换
        memcpy(this, &v, 4);
    }
    void hton() {
        uint32_t v;
        memcpy(&v, this, 4);
        v = htonl(v);
        memcpy(this, &v, 4);
    }
};

// ---------- data3：1 字节位域 ----------
// 声明顺序 bit0 起：lo(3) → hi(5)；单字节无需端序转换
struct Data3 {
    unsigned int lo : 3;         // bit0 ~ 2
    unsigned int hi : 5;         // bit3 ~ 7
    void ntoh() {}               // 空操作
    void hton() {}
};

// ---------- data4：3 字节位域（跨字节）----------
// infoC[8:0]=9, infoB[16:9]=8, infoA[23:17]=7
struct Data4 {
    unsigned int infoC : 9;
    unsigned int infoB : 8;
    unsigned int infoA : 7;

    void ntoh() {                // 3 字节首尾交换（对称，hton 同）
        uint8_t b[3];
        memcpy(b, this, 3);
        uint8_t t = b[0]; b[0] = b[2]; b[2] = t;
        memcpy(this, b, 3);
    }
    void hton() { ntoh(); }
};

// 线上 16 字节：1+2+3+2+4+1+3（不要用 sizeof 当线上长度）
constexpr size_t kUdpPacketWireSize = 16;

struct UdpPacket {
    uint8_t  type;
    uint16_t len;
    uint8_t  fc[3];
    Data1    data1;
    Data2    data2;
    Data3    data3;
    Data4    data4;

    void ntoh() {
        len = ntohs(len);
        data1.ntoh(); data2.ntoh(); data3.ntoh(); data4.ntoh();
    }
    void hton() {
        data1.hton(); data2.hton(); data3.hton(); data4.hton();
        len = htons(len);
    }
};

#pragma pack(pop)

文件 2 — 收包解析

C++parse_packet.cpp
#include "protocol.h"

bool parsePacket(const uint8_t* buf, size_t n, UdpPacket& pkt) {
    if (n < kUdpPacketWireSize) return false;
    size_t i = 0;
    memcpy(&pkt.type,  buf + i, 1); i += 1;
    memcpy(&pkt.len,   buf + i, 2); i += 2;
    memcpy( pkt.fc,    buf + i, 3); i += 3;
    memcpy(&pkt.data1, buf + i, 2); i += 2;
    memcpy(&pkt.data2, buf + i, 4); i += 4;
    memcpy(&pkt.data3, buf + i, 1); i += 1;   // 1 字节
    memcpy(&pkt.data4, buf + i, 3);           // 只拷 3 字节
    pkt.ntoh();                               // 之后直接读位域成员
    return true;
}

文件 3 — 组包 / 发包

C++build_packet.cpp
#include "protocol.h"
#include <vector>

// 填主机序字段 → hton → 按线上顺序拼字节（可直接 UDP sendto）
std::vector<uint8_t> buildPacket(UdpPacket pkt) {
    pkt.len = 3 + 2 + 4 + 1 + 3;              // fc(3) + 尾部 10
    pkt.hton();                               // 转网络序（改的是副本）

    std::vector<uint8_t> out;
    out.reserve(kUdpPacketWireSize);
    auto append = [&](const void* p, size_t n) {
        const auto* b = static_cast<const uint8_t*>(p);
        out.insert(out.end(), b, b + n);
    };
    append(&pkt.type, 1);
    append(&pkt.len, 2);                      // 已 hton，内存映像即线上字节
    append(pkt.fc, 3);
    append(&pkt.data1, 2);
    append(&pkt.data2, 4);
    append(&pkt.data3, 1);
    append(&pkt.data4, 3);
    return out;
}

 *   UdpPacket pkt{};
 *   pkt.type = 0x11;
 *   pkt.fc[0]=0x51; pkt.fc[1]=0x73; pkt.fc[2]=0xB8;
 *   pkt.data1 = { .code=341, .S3=1, .S2=0, .S1=1, .typeCode=5 };
 *   pkt.data2 = { .feild_1=3, .feild_2=9, .feild_3=5, .feild_4=6, .sysCode=0x1234 };
 *   pkt.data3 = { .lo=5, .hi=21 };           // → 线上 AD
 *   pkt.data4 = { .infoC=0x155, .infoB=0xAA, .infoA=0x55 }; // → AB 55 55
 *   auto wire = buildPacket(pkt);
 *   // 11 00 0D 51 73 B8 55 B5 93 65 34 12 AD AB 55 55
 */
口诀：定布局（位域）→ 收：memcpy + ntoh → 读字段；发：填字段 + hton + 拼字节。 动手改字段、模拟发送请打开 交互实验台（默认数据与本文示例一致）。
